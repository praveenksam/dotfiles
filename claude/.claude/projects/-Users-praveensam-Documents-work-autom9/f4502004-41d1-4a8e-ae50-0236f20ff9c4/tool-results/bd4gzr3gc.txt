import asyncio
import random

import httpx

_ANTHROPIC_MESSAGES_URL = "https://api.anthropic.com/v1/messages"
_ANTHROPIC_VERSION = "2023-06-01"
_DEFAULT_MODEL = "claude-sonnet-4-6"
_DEFAULT_MAX_TOKENS = 4096
_MAX_RETRIES = 3
_TRANSIENT_STATUS_CODES = {429, 500, 502, 503, 504, 529}


class _ProviderHTTPError(Exception):
    def __init__(self, status_code: int, body: str) -> None:
        self.status_code = status_code
        super().__init__(f"HTTP {status_code}: {body}")


def _is_transient(exc: Exception) -> bool:
    if isinstance(exc, httpx.TimeoutException):
        return True
    if isinstance(exc, httpx.TransportError):
        return True
    if isinstance(exc, _ProviderHTTPError) and exc.status_code in _TRANSIENT_STATUS_CODES:
        return True
    return False


def _backoff_wait(attempt: int) -> float:
    """Exponential backoff with full jitter: [0, 2^attempt) seconds."""
    return random.uniform(0, 2 ** attempt)


def _split_system_messages(messages: list[dict]) -> tuple[str | None, list[dict]]:
    """
    Extract system-role messages into a top-level system string.
    Anthropic does NOT accept {"role": "system", ...} in the messages array.
    """
    system_parts = []
    non_system = []
    for m in messages:
        if m.get("role") == "system":
            system_parts.append(m.get("content", ""))
        else:
            non_system.append(m)
    system = "\n".join(system_parts) if system_parts else None
    return system, non_system


def _tools_to_anthropic(tools: list[dict]) -> list[dict]:
    """
    Convert internal tool definitions to Anthropic tool format.
    Internal format uses 'parameters'; Anthropic requires 'input_schema'.
    """
    return [
        {
            "name": t["name"],
            "description": t.get("description", ""),
            "input_schema": t.get("parameters", {"type": "object", "properties": {}}),
        }
        for t in tools
    ]


def _parse_content(content: list[dict]) -> tuple[str, list[dict]]:
    """
    Parse Anthropic response content blocks into (text, tool_calls).
    Tool call 'input' is already a dict — no JSON parsing needed.
    Normalizes to internal format: {id, name, arguments: dict}.
    """
    text_parts = []
    tool_calls = []
    for block in content:
        if block.get("type") == "text":
            text_parts.append(block.get("text", ""))
        elif block.get("type") == "tool_use":
            tool_calls.append({
                "id": block.get("id", ""),
                "name": block.get("name", ""),
                "arguments": block.get("input", {}),
            })
    return "\n".join(text_parts), tool_calls


def _normalize_finish_reason(stop_reason: str | None) -> str:
    mapping = {
        "end_turn": "stop",
        "tool_use": "tool_calls",
        "max_tokens": "length",
        "stop_sequence": "stop",
    }
    return mapping.get(stop_reason or "", stop_reason or "stop")


class AnthropicProvider(LLMProvider):
    """Concrete LLMProvider implementation for the Anthropic Messages API."""

    def __init__(
        self,
        api_key: str,
        model: str = _DEFAULT_MODEL,
        max_tokens: int = _DEFAULT_MAX_TOKENS,
    ) -> None:
        self._api_key = api_key
        self._model = model
        self._max_tokens = max_tokens

    async def complete(self, messages: list[dict], tools: list[dict]) -> LLMResponse:
        """
        Call Anthropic Messages API.
        Retries on transient errors with exponential backoff + jitter (ARC-10).
        Raises ProviderError after 3 retries.
        """
        system, filtered_messages = _split_system_messages(messages)

        payload: dict = {
            "model": self._model,
            "max_tokens": self._max_tokens,
            "messages": filtered_messages,
        }
        if system:
            payload["system"] = system
        if tools:
            payload["tools"] = _tools_to_anthropic(tools)

        last_exc: Exception | None = None
        for attempt in range(_MAX_RETRIES + 1):
            try:
                return await self._call_api(payload)
            except Exception as exc:
                if not _is_transient(exc):
                    raise ProviderError("anthropic", str(exc)) from exc
                last_exc = exc
                if attempt < _MAX_RETRIES:
                    await asyncio.sleep(_backoff_wait(attempt))

        raise ProviderError("anthropic", f"Max retries exceeded: {last_exc}") from last_exc

    async def _call_api(self, payload: dict) -> LLMResponse:
        async with httpx.AsyncClient(timeout=30.0) as client:
            response = await client.post(
                _ANTHROPIC_MESSAGES_URL,
                headers={
                    "x-api-key": self._api_key,
                    "anthropic-version": _ANTHROPIC_VERSION,
                    "content-type": "application/json",
                },
                json=payload,
            )
        if response.status_code != 200:
            raise _ProviderHTTPError(response.status_code, response.text)
        data = response.json()
        text, tool_calls = _parse_content(data.get("content", []))
        return LLMResponse(
            content=text,
            tool_calls=tool_calls,
            finish_reason=_normalize_finish_reason(data.get("stop_reason")),
        )

class _ProviderHTTPError(Exception):
    def __init__(self, status_code: int, body: str) -> None:

def _is_transient(exc: Exception) -> bool:
    if isinstance(exc, httpx.TimeoutException):
        return True
    if isinstance(exc, httpx.TransportError):
        return True
    if isinstance(exc, _ProviderHTTPError) and exc.status_code in _TRANSIENT_STATUS_CODES:

    return False


def _backoff_wait(attempt: int) -> float:
    """Exponential backoff with full jitter: [0, 2^attempt) seconds."""
    return random.uniform(0, 2 ** attempt)


def _tools_to_openai(tools: list[dict]) -> list[dict]:
    """Convert internal tool definitions to OpenAI function tool format."""
    return [
        {
            "type": "function",
            "function": {
                "name": t["name"],
                "description": t.get("description", ""),
                "parameters": t.get("parameters", {}),
            },
        }
        for t in tools
    ]


def _parse_tool_calls(raw_calls: list[dict] | None) -> list[dict]:
    """
    Normalize OpenAI tool calls to internal format: {id, name, arguments: dict}.
    OpenAI sends arguments as a JSON string — parse with json.loads().
    """
    if not raw_calls:
        return []
    result = []
    for tc in raw_calls:
        fn = tc.get("function", {})
        args_raw = fn.get("arguments", "{}")
        try:
            args = json.loads(args_raw)
        except json.JSONDecodeError:
            args = {}
        result.append({
            "id": tc.get("id", ""),
            "name": fn.get("name", ""),
            "arguments": args,
        })
    return result


def _normalize_finish_reason(reason: str | None) -> str:
    mapping = {
        "stop": "stop",
        "tool_calls": "tool_calls",
        "length": "length",
        "content_filter": "content_filter",
    }
    return mapping.get(reason or "", reason or "stop")


class OpenAIProvider(LLMProvider):
    """Concrete LLMProvider implementation for the OpenAI Chat Completions API."""

    def __init__(self, api_key: str, model: str = _DEFAULT_MODEL) -> None:
        self._api_key = api_key
        self._model = model

    async def complete(self, messages: list[dict], tools: list[dict]) -> LLMResponse:
        """
        Call OpenAI Chat Completions API.
        Retries on transient errors with exponential backoff + jitter (ARC-10).
        Raises ProviderError after 3 retries.
        """
        payload: dict = {
            "model": self._model,
            "messages": messages,
        }
        if tools:
            payload["tools"] = _tools_to_openai(tools)

        last_exc: Exception | None = None
        for attempt in range(_MAX_RETRIES + 1):
            try:
                return await self._call_api(payload)
            except Exception as exc:
                if not _is_transient(exc):
                    raise ProviderError("openai", str(exc)) from exc
                last_exc = exc
                if attempt < _MAX_RETRIES:
                    await asyncio.sleep(_backoff_wait(attempt))

        raise ProviderError("openai", f"Max retries exceeded: {last_exc}") from last_exc

    async def _call_api(self, payload: dict) -> LLMResponse:
        async with httpx.AsyncClient(timeout=30.0) as client:
            response = await client.post(
                _OPENAI_COMPLETIONS_URL,
                headers={
                    "Authorization": f"Bearer {self._api_key}",
                    "Content-Type": "application/json",
                },
                json=payload,
            )
        if response.status_code != 200:
            raise _ProviderHTTPError(response.status_code, response.text)
        data = response.json()
        choice = data["choices"][0]
        message = choice["message"]
        return LLMResponse(
            content=message.get("content") or "",
            tool_calls=_parse_tool_calls(message.get("tool_calls")),
            finish_reason=_normalize_finish_reason(choice.get("finish_reason")),
        )

_ORG_ID = "default"

_PROVIDER_MAP: dict[str, type[LLMProvider]] = {
    "openai": OpenAIProvider,
    "anthropic": AnthropicProvider,
}


async def get_provider(db: AsyncSession) -> LLMProvider:
    """
    Resolve the active LLM provider from DB credentials.
    Queries DB on every call — supports provider switching without app restart (FR28).
    Raises ProviderError if no active credential exists.
    """
    result = await db.execute(
        select(CredentialStore).where(
            CredentialStore.org_id == _ORG_ID,
            CredentialStore.is_active.is_(True),
        )
    )
    row = result.scalar_one_or_none()
    if row is None:
        raise ProviderError(
            "none",
            "No active LLM provider configured. Set credentials via admin panel.",
        )

    provider_cls = _PROVIDER_MAP.get(row.provider)
    if provider_cls is None:
        raise ProviderError(row.provider, f"Unknown provider '{row.provider}'")

    api_key = decrypt(row.encrypted_key)
    return provider_cls(api_key=api_key)
